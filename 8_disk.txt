
#include <stdio.h>
#include <stdlib.h>

void SSTF(int arr[], int n, int head)
{
    int visited[100] = {0};
    int total = 0, current = head;

    printf("\nSSTF Seek Sequence: %d", head);

    for (int i = 0; i < n; i++)
    {
        int min = 9999, index = -1;

        // Find the nearest unvisited request
        for (int j = 0; j < n; j++)
        {
            if (!visited[j])
            {
                int diff = abs(current - arr[j]);

                if (diff < min)
                {
                    min = diff;
                    index = j;
                }
            }
        }

        visited[index] = 1;
        total += abs(current - arr[index]);
        current = arr[index];

        printf(" -> %d", current);
    }

    printf("\nTotal Head Movement = %d\n", total);
}

void SCAN(int arr[], int n, int head, int diskSize)
{
    int a[100], total = 0;
    int current = head;

    // Copy requests into another array
    for (int i = 0; i < n; i++)
        a[i] = arr[i];

    // Sort requests in ascending order
    for (int i = 0; i < n - 1; i++)
    {
        for (int j = 0; j < n - i - 1; j++)
        {
            if (a[j] > a[j + 1])
            {
                int temp = a[j];
                a[j] = a[j + 1];
                a[j + 1] = temp;
            }
        }
    }

    printf("\nSCAN Seek Sequence: %d", head);

    // Move towards higher cylinders (away from spindle)
    for (int i = 0; i < n; i++)
    {
        if (a[i] >= head)
        {
            total += abs(current - a[i]);
            current = a[i];
            printf(" -> %d", current);
        }
    }

    // Move to the last cylinder
    if (current != diskSize - 1)
    {
        total += abs(current - (diskSize - 1));
        current = diskSize - 1;
        printf(" -> %d", current);
    }

    // Reverse direction and service remaining requests
    for (int i = n - 1; i >= 0; i--)
    {
        if (a[i] < head)
        {
            total += abs(current - a[i]);
            current = a[i];
            printf(" -> %d", current);
        }
    }

    printf("\nTotal Head Movement = %d\n", total);
}

void CLOOK(int arr[], int n, int head)
{
    int a[100], total = 0;
    int current = head;

    // Copy requests
    for (int i = 0; i < n; i++)
        a[i] = arr[i];

    // Sort in ascending order
    for (int i = 0; i < n - 1; i++)
    {
        for (int j = 0; j < n - i - 1; j++)
        {
            if (a[j] > a[j + 1])
            {
                int temp = a[j];
                a[j] = a[j + 1];
                a[j + 1] = temp;
            }
        }
    }

    printf("\nC-LOOK Seek Sequence: %d", head);

    // First service requests towards higher cylinders
    for (int i = 0; i < n; i++)
    {
        if (a[i] >= head)
        {
            total += abs(current - a[i]);
            current = a[i];
            printf(" -> %d", current);
        }
    }

    // Jump to the lowest request
    for (int i = 0; i < n; i++)
    {
        if (a[i] < head)
        {
            total += abs(current - a[i]);
            current = a[i];
            printf(" -> %d", current);
        }
    }

    printf("\nTotal Head Movement = %d\n", total);
}

int main()
{
    int n, head, diskSize;
    int arr[100];

    printf("Enter number of requests: ");
    scanf("%d", &n);

    printf("Enter request queue:\n");
    for (int i = 0; i < n; i++)
        scanf("%d", &arr[i]);

    printf("Enter initial head position: ");
    scanf("%d", &head);

    printf("Enter disk size: ");
    scanf("%d", &diskSize);

    SSTF(arr, n, head);
    SCAN(arr, n, head, diskSize);
    CLOOK(arr, n, head);

    return 0;
}
